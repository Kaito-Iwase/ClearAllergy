async function one(id, change, expected, cancel = false) {
  const context = await browser.newContext({ viewport: { width: 390, height: 900 }, timezoneId: 'Asia/Tokyo' });
  await context.route('**/*', route => {
    if (new URL(route.request().url()).origin === origin) return route.continue();
    result.blockedExternalRequests.push(route.request().url());
    return route.abort();
  });
  const page = await context.newPage();
  page.setDefaultTimeout(15000);
  const dialogs = [], puts = [];
  page.on('pageerror', error => result.errors.push({ id, message: error.message }));
  page.on('dialog', async dialog => { dialogs.push({ type: dialog.type(), message: dialog.message(), accepted: !cancel }); await (cancel ? dialog.dismiss() : dialog.accept()); });
  page.on('request', request => { if (request.method() === 'PUT') puts.push(JSON.parse(request.postData())); });
  try {
    await page.goto(origin + '/admin/menus/trial-dish-one/edit');
    await page.getByLabel('🧺 原材料名', { exact: true }).waitFor();
    await page.getByText('保存済みの公開設定：公開', { exact: false }).waitFor();
    const before = await (await context.request.get(origin + '/__trial__/state')).json();
    assert.equal(before.menu.isPublished, true);
    await change(page);
    const expectedButton = await page.getByRole('button', { name: /保存時(?:は|に)/ }).innerText();
    await page.screenshot({ path: path.join(output, id + '-before.png'), fullPage: true });
    let response;
    if (cancel) {
      await page.getByRole('button', { name: '保存する', exact: true }).click();
      await page.waitForTimeout(200);
      assert.equal(puts.length, 0);
    } else {
      const waiting = page.waitForResponse(r => r.request().method() === 'PUT' && r.url() === origin + '/api/admin/menus/trial-dish-one');
      await page.getByRole('button', { name: '保存する', exact: true }).click();
      const r = await waiting; response = { status: r.status(), body: await r.json() };
      await page.getByText('保存しました。', { exact: true }).waitFor();
      assert.equal(response.status, 200);
      assert.equal(puts.length, 1);
      assert.equal(puts[0].isPublished, expected.published);
      assert.equal(response.body.menu.isPublished, expected.published);
    }
    const after = await (await context.request.get(origin + '/__trial__/state')).json();
    assert.equal(dialogs.length, expected.dialogs);
    const saysPublish = dialogs.some(d => d.message.includes('公開する設定で'));
    assert.equal(saysPublish, expected.saysPublish);
    if (cancel) assert.deepEqual(after.menu, before.menu);
    const mismatch = !cancel && saysPublish && puts[0].isPublished === false;
    assert.equal(mismatch, expected.mismatch);
    await page.screenshot({ path: path.join(output, id + '-after.png'), fullPage: true });
    const row = { id, expectedButton, dialogs, requests: puts, response, finalPublicSetting: after.menu.isPublished, finalReviewState: after.listRow.foodReviewState, finalPublicVisible: after.publicVisible, mismatch, status: 'PASS' };
    result.cases.push(row); console.log('PASS ' + id + ': mismatch=' + mismatch + ', dialogs=' + dialogs.length + ', PUT=' + puts.length);
  } finally { await context.close(); }
}
